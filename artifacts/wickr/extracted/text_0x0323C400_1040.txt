CREATE TABLE Wickr_Message (_id INTEGER PRIMARY KEY, state INTEGER DEFAULT 0, msgID text, vGroupID TEXT, class INTEGER, readTimestamp INTEGER DEFAULT 0, timestamp INTEGER, usec INTEGER DEFAULT 0, type TEXT, convo INTEGER, destructTime INTEGER, appSender INTEGER, userSender INTEGER, senderAppID TEXT, senderUserID TEXT, bodyKey BLOB, bodyText STRING, bodyData BLOB, tMsgID STRING, ttl INTEGER, needsSync INTEGER DEFAULT 0, isRead INTEGER DEFAULT 0, bor INTEGER DEFAULT -1, isMsgStarred INTEGER DEFAULT 0, msgFileName TEXT DEFAULT NULL, skipCleanup INTEGER DEFAULT 0, clientMSN INTEGER DEFAULT 0, isHidden INTEGER DEFAULT 0, retryCount INTEGER DEFAULT 0, mention INTEGER DEFAULT 0, isUnreadError INTEGER DEFAULT 0, rrTimestamp INTEGER DEFAULT 0, rrReceived INTEGER DEFAULT 0, isNetworkError INTEGER DEFAULT 0, isFileUploadError INTEGER DEFAULT 0, lastSendAttemptTimestamp INTEGER DEFAULT 0, FOREIGN KEY(convo) REFERENCES Wickr_Convo(_id), FOREIGN KEY(appSender) REFERENCES Wickr_App(_id), FOREIGN KEY(userSender) REFERENCES Wickr_User(_id));
