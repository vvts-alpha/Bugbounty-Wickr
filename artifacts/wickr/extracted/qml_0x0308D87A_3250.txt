�import QtQuick
import onboardStatus 1.0
import "../Commons/"
import "../widgets/"
import QtQuick.Controls.Basic

BaseComponent {
    id: root
    signal manualInviteEntry(string email)

    backVisible: false

    headerText: qsTr("Check your email")

    onBackButtonTabPressed: {
        if (resendButton.visible)
            resendButton.focus = true
    }
    onBackButtonReturnPressed: {
        stateHelper.backClicked()
    }
    onBackButtonBackTabPressed: {
        if (resendButton.visible)
            resendButton.focus = true
    }

    function resendPressed() {
        manualInviteEntry(stateHelper.email);
        stateHelper.nextState(OnboardStatus.CheckYourEmail)

    }

    Item {
        id: textItem

        anchors {
            top: parent.top
            topMargin: 88 * ScreenRM.scale
            horizontalCenter: parent.horizontalCenter
        }
        height: 144 * ScreenRM.scale
        width: 420 * ScreenRM.scale

        Text {
            anchors.fill: parent
            text: qsTr("We’ve sent an email to you at %1. It may take a few minutes to arrive. Use the link to verify your account, then return here.").arg("<b>" + stateHelper.email + "</b>") + "<br><br>" + qsTr("If it doesn’t arrive within 30 minutes, contact your administrator.")
            font.pointSize: 18 * ScreenRM.fontScale
            font.family: fonts.osReg
            font.weight: Font.Normal
            color: Branding.palette_gbl_p11
            horizontalAlignment: Text.AlignHCenter
            wrapMode: Text.Wrap
            textFormat: Text.RichText
        }
    }

    WrappedImage {
        id: emailLargeImage

        anchors {
            top: textItem.bottom
            topMargin: 52 * ScreenRM.scale
            horizontalCenter: parent.horizontalCenter
        }
        height: 141 * ScreenRM.scale
        width: 141 * ScreenRM.scale
        source: "qrc:/images/ic_email.png"
    }

    WickrIconTextButton {
        id: resendButton

        anchors {
            top: emailLargeImage.bottom
            topMargin: 30 * ScreenRM.scale
            horizontalCenter: parent.horizontalCenter
        }
        styleName: "standardLarge"
        text: "Resend email"
        textFont.weight: Font.Normal
        width: standardWidth
        height: standardHeight
        focus: true
        mainBackgroundColor: "transparent"
        innerRectColor: "transparent"

        // Keep this button invisible for now. I need to investigate this functionality.
        //visible: !isSSOVerify
        visible: false

        onClicked: {
            resendPressed();
        }

        onMouseReleased: stateHelper.nextState(OnboardStatus.CheckYourEmail)
        Keys.onTabPressed: {
            backButton.focus = true
        }
        Keys.onBacktabPressed: {
            backButton.focus = true
        }
        Keys.onReturnPressed: {
            resendPressed();
        }
    }
    Component.onCompleted: {
        if (resendButton.visible) {
            resendButton.focus = true
            resendButton.forceActiveFocus()
        } else {
            backButton.focus = true
        }
    }
}
